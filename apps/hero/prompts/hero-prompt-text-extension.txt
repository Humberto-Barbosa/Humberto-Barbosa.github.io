        }

        function escapeHTML(text) {
            return text
                .replaceAll("&", "&amp;")
                .replaceAll("<", "&lt;")
                .replaceAll(">", "&gt;")
                .replaceAll('"', "&quot;")
                .replaceAll("'", "&#39;");
        }

        function generateHTML() {
            const headingText = escapeHTML(heading.innerText);
            const description = document.querySelector(
                'meta[name="description"]'

            let subheadingHTML = "";

            if (settings.showSubheading) {

</body>
</html>`;
        }

        function refreshOutput() {
            copyStatus.textContent = "";

            if (!manualCopy.hidden) {
                htmlOutput.value = generateHTML();
            }
        }

        async function copyHTML() {
            const html = generateHTML();

            try {
                if (!navigator.clipboard?.writeText) {
                    throw new Error("Clipboard API unavailable");
                }

                await navigator.clipboard.writeText(html);

                manualCopy.hidden = true;
                copyStatus.textContent =
                    "HTML copied—paste it into VS Code.";
            } catch (error) {
                manualCopy.hidden = false;
                htmlOutput.value = generateHTML();

                manualCopy.scrollIntoView({
                    behavior: "smooth",
                    block: "center"
                });

                htmlOutput.focus({ preventScroll: true });
                htmlOutput.select();

                copyStatus.textContent =
                    "Copy the selected HTML manually.";
            }
        }

        function insertPlainText(element, text) {
            const selection = window.getSelection();

            if (!selection || selection.rangeCount === 0) {
                return;
            }

            const range = selection.getRangeAt(0);

            if (!element.contains(range.commonAncestorContainer)) {
                return;
            }

            range.deleteContents();

            const textNode = document.createTextNode(text);
            range.insertNode(textNode);
            range.setStartAfter(textNode);
            range.collapse(true);

            selection.removeAllRanges();
            selection.addRange(range);

            refreshOutput();
        }

        function handlePaste(event) {
            event.preventDefault();

            const text = event.clipboardData.getData("text/plain");
            insertPlainText(event.currentTarget, text);
        }

            event.preventDefault();
        }
